// Smoothed posture state (prevents 30fps landmark flicker from blocking clicks) let smoothFaceRatio = null; let smoothPitch = 0; let smoothYaw = 0; let smoothRoll = 0; let smoothFacePixelW = null; let lastValidTimestamp = 0; function checkPostureAndDistance(landmarks, w, h) { const p127 = landmarks[127]; const p356 = landmarks[356]; const p10 = landmarks[10]; const p152 = landmarks[152]; const p1 = landmarks[1]; const p33 = landmarks[33]; const p263 = landmarks[263]; const rawFacePixelWidth = Math.hypot((p356.x - p127.x) * w, (p356.y - p127.y) * h); const rawFaceRatio = rawFacePixelWidth / w; const faceHeightNorm = Math.hypot(p152.x - p10.x, p152.y - p10.y) || 0.1; const rawPitch = (p10.z - p152.z) / faceHeightNorm; const eyeMidX = (p33.x + p263.x) / 2; const eyeSpanX = Math.abs(p263.x - p33.x) || 0.05; const rawYaw = Math.abs(p1.x - eyeMidX) / eyeSpanX; const eyeDx = (p263.x - p33.x) * w; const eyeDy = (p263.y - p33.y) * h; const rawRoll = Math.abs(Math.atan2(eyeDy, eyeDx) * 180 / Math.PI); // Exponential moving average (alpha = 0.25) eliminates frame-to-frame jitter const a = 0.25; smoothFacePixelW = smoothFacePixelW === null ? rawFacePixelWidth : smoothFacePixelW * (1 - a) + rawFacePixelWidth * a; smoothFaceRatio = smoothFaceRatio === null ? rawFaceRatio : smoothFaceRatio * (1 - a) + rawFaceRatio * a; smoothPitch = smoothPitch * (1 - a) + rawPitch * a; smoothYaw = smoothYaw * (1 - a) + rawYaw * a; smoothRoll = smoothRoll * (1 - a) + rawRoll * a; const estDistanceCm = Math.round(9.2 / smoothFaceRatio); // Natural, stable tolerances (~45–75 cm, ±16% pitch, ±18% yaw, ±9° roll) const isTooClose = smoothFaceRatio > 0.205; const isTooFar = smoothFaceRatio < 0.122; const isChinUp = smoothPitch > 0.16; const isChinDown = smoothPitch < -0.16; const isTurned = smoothYaw > 0.18; const isTilted = smoothRoll > 9.0; const instantValid = !isTooClose && !isTooFar && !isChinDown && !isChinUp && !isTurned && !isTilted; if (instantValid) { lastValidTimestamp = performance.now(); } // 450ms grace window so reaching for the mouse/tap never locks the button mid-click const isValid = instantValid || (performance.now() - lastValidTimestamp < 450); return { facePixelWidth: smoothFacePixelW, estDistanceCm, isTooClose, isTooFar, isChinDown, isChinUp, isTurned, isTilted, isValid }; }
